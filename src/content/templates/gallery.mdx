<%*
const columns = await tp.system.suggester(
  ["2 columns", "3 columns", "4 columns"],
  [2, 3, 4],
);
const figures = [];
let file;
while (
  columns &&
  (file = await tp.system.prompt(
    `Image ${figures.length + 1} filename (blank to finish)`,
    "",
  ))
) {
  const alt = (await tp.system.prompt("Alt text", "")) || "";
  const caption = (await tp.system.prompt("Caption or credit (optional)", "")) || "";
  const imageName = `galleryImage${figures.length + 1}`;
  figures.push({ file, alt, caption, imageName });
}
if (figures.length) {
  const imports = figures
    .map(({ file, imageName }) => `import ${imageName} from "./${file}";`)
    .join("\n");
  const content = figures
    .map(
      ({ alt, caption, imageName }) =>
        `  <Figure src={${imageName}} alt={${JSON.stringify(alt)}}>${caption ? `\n    ${caption}\n  ` : ""}</Figure>`,
    )
    .join("\n");
  tR += `import Figure from "../../../components/content/Figure.astro";\nimport Gallery from "../../../components/content/Gallery.astro";\n${imports}\n\n<Gallery columns={${columns}}>\n${content}\n</Gallery>`;
}
-%>
